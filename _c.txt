const { scene, Axes, Dot, Line, Arrow, Text, Create, FadeIn, VGroup, params } = ctx;
// 定义函数 f(x) = x^2
const f = (x) => x * x;
// 导数 f'(x) = 2*x
const df = (x) => 2 * x;

const x = params.x; // 使用参数

// 创建坐标轴（蓝色系）
const axes = new Axes({
    xRange: [-3.5, 3.5, 1],
    yRange: [-0.5, 9.5, 1],
    xLength: 8,
    yLength: 5,
    axisConfig: { color: "#2b3a52", strokeWidth: 2 }
});

// 画出曲线 y = x^2
const curve = axes.plot(f, { xRange: [-3.5, 3.5], color: "#4a9eff", strokeWidth: 3 });

// 当前点的坐标
const px = x;
const py = f(px);
const pointPos = axes.c2p(px, py);

// 当前点
const dot = new Dot({ point: pointPos, radius: 0.08, color: "#5fb0ff" });

// 梯度箭头：从当前点指向 (px + 0.5, py + 0.5 * df(px))
const gradDir = df(px);
const dx = 0.4; // 箭头水平长度
const endX = px + dx;
const endY = py + dx * gradDir;
const arrowEnd = axes.c2p(endX, endY);
const arrow = new Arrow({
    start: pointPos,
    end: arrowEnd,
    color: "#5fb0ff",
    strokeWidth: 3
});

// 标注文本：梯度方向
const label = new Text({
    text: "梯度方向",
    fontSize: 0.3,
    color: "#a8c8e8",
    fontFamily: '"Microsoft YaHei","PingFang SC",sans-serif'
});
label.moveTo(arrowEnd).shift([0.2, 0.2, 0]);

// 可选的辅助虚线（从当前点水平或垂直，可选）
const hLine = new Line({
    start: pointPos,
    end: axes.c2p(px, 0),
    color: "#9aa6b8",
    strokeWidth: 1.5,
    dashArray: [0.05, 0.05]
});

// 分组：点、箭头、标注、辅助线
const group = new VGroup(dot, arrow, label, hLine);

// 动画步骤
scene.add(axes);
await scene.play(new Create(curve));
await scene.play(new FadeIn(group));
