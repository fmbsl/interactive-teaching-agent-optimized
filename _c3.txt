const { scene, Axes, Dot, Line, Text, Arrow, Create, FadeIn, VGroup } = ctx;

// 创建坐标轴和曲线（模拟山坡截面）
const axes = new Axes({
  xRange: [-3.5, 4, 1],
  yRange: [0, 4.5, 1],
  xLength: 10,
  yLength: 6,
  axisConfig: { color: "#2b3a52", strokeWidth: 2 }
});
const curve = axes.plot((x) => 0.3 * (x - 0.5) ** 2 + 0.5, {
  xRange: [-2.5, 3.5],
  color: "#4a9eff",
  strokeWidth: 3
});

// 定义下山路径上的关键点（从左侧高点→谷底）
const points = [
  { x: -2,   label: "起点" },
  { x: -1.5, label: "" },
  { x: -1,   label: "" },
  { x: -0.5, label: "" },
  { x: 0,    label: "" },
  { x: 0.5,  label: "谷底" }
];

const dots = points.map((p, i) => {
  const y = 0.3 * (p.x - 0.5) ** 2 + 0.5;
  const dot = new Dot({
    point: axes.c2p(p.x, y),
    radius: i === 0 || i === points.length - 1 ? 0.1 : 0.06,
    color: i === 0 ? "#5fb0ff" : i === points.length - 1 ? "#4a9eff" : "#a8c8e8"
  });
  return dot;
});

// 箭头连接相邻点（指示最陡方向）
const arrows = [];
for (let i = 0; i < dots.length - 1; i++) {
  const start = dots[i].getCenter();
  const end = dots[i + 1].getCenter();
  const arrow = new Arrow({
    start,
    end,
    color: "#9aa6b8",
    strokeWidth: 2
  });
  arrows.push(arrow);
}

// 标注文字
const title = new Text({
  text: "下山找最低点",
  fontSize: 0.4,
  color: "#a8c8e8",
  fontFamily: '"Microsoft YaHei","PingFang SC",sans-serif'
});
title.moveTo([0, 3.2, 0]);

const startLabel = new Text({
  text: "当前点",
  fontSize: 0.25,
  color: "#5fb0ff",
  fontFamily: '"Microsoft YaHei","PingFang SC",sans-serif'
});
startLabel.moveTo(dots[0].getCenter().concat([0, 0.3, 0]));

const endLabel = new Text({
  text: "最低点",
  fontSize: 0.25,
  color: "#4a9eff",
  fontFamily: '"Microsoft YaHei","PingFang SC",sans-serif'
});
endLabel.moveTo(dots[dots.length - 1].getCenter().concat([0, -0.3, 0]));

// 分组路径元素
const pathGroup = new VGroup(...dots, ...arrows, startLabel, endLabel);

// 第一步：显示坐标轴和曲线
scene.add(axes, curve);
await scene.play(new Create(curve));

// 第二步：显示标题
await scene.play(new FadeIn(title));

// 第三步：逐步显示下山路径（带方向）
await scene.play(new FadeIn(pathGroup));
